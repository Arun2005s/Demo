
using GroceryShopApp.Model;
using GroceryShopApp.Services;

namespace GroceryShopApp.View
{
    internal class ProductView
    {
        private readonly ProductServices _productServices;

        public ProductView(ProductServices productServices)
        {
            this._productServices = productServices;
        }

        internal void DisplayProductMenu()
        {
            while (true)
            {
                Console.Write(@"
==================================
          PRODUCT MENU
==================================
1. Vegetables
2. Fruits
3. Cookware
4. Back to Main Menu

Enter Choice: ");

                string? choice = Console.ReadLine();

                if (!int.TryParse(choice, out int selectedCategory))
                {
                    ConsoleHelper.WriteErrorMessage(
                        "\nInvalid selection.\n");

                    continue;
                }

                if (selectedCategory == 4)
                {
                    Console.Clear();
                    return;
                }

                if (!Enum.IsDefined(
                    typeof(Categories), selectedCategory))
                {
                    ConsoleHelper.WriteErrorMessage(
                        "\nInvalid category.\n");

                    continue;
                }

                Categories category =
                    (Categories)selectedCategory;

                Console.Clear();

                this.DisplayProducts(category);
            }
        }

        private void DisplayProducts(Categories category)
        {
            List<Product> products =
                this._productServices.GetProductsByCategory(category);

            while (true)
            {
                Console.WriteLine($@"
==================================
       {category.ToString().ToUpper()}
==================================");

                this.DisplayProductList(products);

                Console.Write(@"
[S] Search Product
[B] Back to Categories

Enter product number to select: ");

                string? choice = Console.ReadLine();

                if (string.Equals(
                    choice, "B",
                    StringComparison.OrdinalIgnoreCase))
                {
                    Console.Clear();
                    return;
                }

                if (string.Equals(
                    choice, "S",
                    StringComparison.OrdinalIgnoreCase))
                {
                    Console.Write(
                        "\nEnter product name to search: ");

                    string searchName =
                        Console.ReadLine() ?? string.Empty;

                    if (string.IsNullOrWhiteSpace(searchName))
                    {
                        ConsoleHelper.WriteErrorMessage(
                            "\nSearch name cannot be empty.\n");

                        continue;
                    }

                    products =
                        this._productServices.SearchProducts(
                            category, searchName.Trim());

                    Console.Clear();

                    if (products.Count == 0)
                    {
                        ConsoleHelper.WriteErrorMessage(
                            "\nNo products found.\n");
                    }

                    continue;
                }

                if (!int.TryParse(choice, out int productNumber) ||
                    productNumber < 1 ||
                    productNumber > products.Count)
                {
                    ConsoleHelper.WriteErrorMessage(
                        "\nInvalid product selection.\n");

                    continue;
                }

                Product selectedProduct =
                    products[productNumber - 1];

                this.DisplaySelectedProduct(selectedProduct);
            }
        }

        private void DisplayProductList(List<Product> products)
        {
            if (products.Count == 0)
            {
                ConsoleHelper.WriteErrorMessage(
                    "\nNo products found.\n");

                return;
            }

            for (int i = 0; i < products.Count; i++)
            {
                Product product = products[i];

                Console.WriteLine(
                    $"\n{i + 1}. Product: {product.Name}");

                Console.WriteLine(
                    $"   Price: Rs.{product.Price}");

                Console.WriteLine(
                    product.Quantity == 0
                    ? "   Status: Out of stock"
                    : $"   Remaining Stock: {product.Quantity}");

                Console.WriteLine(
                    product.Quantity == 0
                    ? "   Add to cart: Unavailable"
                    : "   [A] Add to cart (Next release)");
            }
        }

        private void DisplaySelectedProduct(Product product)
        {
            Console.WriteLine("\n===== PRODUCT DETAILS =====");

            Console.WriteLine($"Name: {product.Name}");
            Console.WriteLine($"Price: Rs.{product.Price}");
            Console.WriteLine($"Category: {product.Category}");
            Console.WriteLine($"Stock: {product.Quantity}");
            Console.WriteLine($"Image: {product.ImageFileName}");

            if (product.Quantity == 0)
            {
                ConsoleHelper.WriteErrorMessage(
                    "\nOut of stock. Cannot add this product.\n");

                return;
            }

            Console.WriteLine(
                "\nAdd to cart functionality will be available in the next release.");
        }
    }
}








namespace GroceryShopApp.View
{
    internal class MainMenu
    {
        private string _currentUser;

        private readonly ProductView _productView;

        public MainMenu(ProductView productView)
        {
            this._currentUser = string.Empty;
            this._productView = productView;
        }

        public void AssignCurrentUser(string userName)
        {
            this._currentUser = userName;
        }

        internal void MenuOptions()
        {
            while (true)
            {
                Console.Write($@"
==================================
            MAIN MENU
==================================
Welcome, {this._currentUser}

[P] Products
[C] Cart Items
[L] Log out

Enter Choice: ");

                ConsoleKey userChoice =
                    Console.ReadKey().Key;

                Console.Clear();

                switch (userChoice)
                {
                    case ConsoleKey.P:

                        this._productView.DisplayProductMenu();

                        break;

                    case ConsoleKey.C:

                        Console.WriteLine(
                            "Cart functionality will be implemented in the next release.");

                        break;

                    case ConsoleKey.L:

                        this._currentUser = string.Empty;

                        ConsoleHelper.WriteColored(
                            "\nLogging out...",
                            ConsoleColor.Yellow);

                        Thread.Sleep(1000);

                        Console.Clear();

                        return;

                    default:

                        ConsoleHelper.WriteErrorMessage(
                            "\nInvalid choice.\n");

                        break;
                }
            }
        }
    }
}
