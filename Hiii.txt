
using System.Text.Json;
using GroceryShopApp.Model;

namespace GroceryShopApp.Repository
{
    /// <summary>
    /// Handles the storage and retrieval of order data to and from files.
    /// </summary>
    internal class OrderRepository
    {
        private readonly List<Order> _orders = new List<Order>();
        private readonly string _filePath;

        private readonly JsonSerializerOptions _options = new JsonSerializerOptions
        {
            WriteIndented = true,
        };

        /// <summary>
        /// Initializes a new instance of the <see cref="OrderRepository"/> class.
        /// </summary>
        /// <param name="filePath">Path where the order data-file is located.</param>
        public OrderRepository(string filePath)
        {
            this._filePath = filePath;
            this.LoadAll();
        }

        /// <summary>
        /// Adds a new order to the repository.
        /// </summary>
        /// <param name="order">Order details.</param>
        internal void AddOrder(Order order)
        {
            this._orders.Add(order);
            this.WriteAll();
        }

        /// <summary>
        /// Retrieves all the orders.
        /// </summary>
        /// <returns>List of all orders.</returns>
        internal List<Order> GetAllOrders()
        {
            return this._orders.ToList();
        }

        /// <summary>
        /// Retrieves orders of a particular user.
        /// </summary>
        /// <param name="userName">Username of the user.</param>
        /// <returns>List of user orders.</returns>
        internal List<Order> GetUserOrders(string userName)
        {
            return this._orders
                .Where(order => order.UserName == userName)
                .ToList();
        }

        private void WriteAll()
        {
            string fileData = JsonSerializer.Serialize(this._orders, this._options);
            File.WriteAllText(this._filePath, fileData);
        }

        private void LoadAll()
        {
            if (!File.Exists(this._filePath))
            {
                this.WriteAll();
                return;
            }

            string fileData = File.ReadAllText(this._filePath);

            if (string.IsNullOrWhiteSpace(fileData))
            {
                return;
            }

            List<Order> orders = JsonSerializer.Deserialize<List<Order>>(fileData, this._options)
                ?? new List<Order>();

            this._orders.AddRange(orders);
        }
    }
}
