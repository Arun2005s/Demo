
using System.Text.Json;
using GroceryShopApp.Model;

namespace GroceryShopApp.Repository
{
    /// <summary>
    /// Handles the storage and retrieval of cart data to and from files.
    /// </summary>
    internal class CartRepository
    {
        private readonly List<Cart> _carts = new List<Cart>();
        private readonly string _filePath;
        private readonly ProductRepository _productRepository;

        private readonly JsonSerializerOptions _options = new JsonSerializerOptions
        {
            WriteIndented = true,
        };

        /// <summary>
        /// Initializes a new instance of the <see cref="CartRepository"/> class.
        /// </summary>
        /// <param name="filePath">Path where the cart data-file is located.</param>
        /// <param name="productRepository">Repository of the products.</param>
        public CartRepository(string filePath, ProductRepository productRepository)
        {
            this._filePath = filePath;
            this._productRepository = productRepository;
            this.LoadAll();
        }

        /// <summary>
        /// Retrieves all cart products of a particular user.
        /// </summary>
        /// <param name="userName">Username of the user.</param>
        /// <returns>List of cart products.</returns>
        internal List<Cart> GetCart(string userName)
        {
            return this._carts.Where(cart => cart.UserName == userName).ToList();
        }

        /// <summary>
        /// Adds a product to the user's cart or updates its quantity.
        /// </summary>
        /// <param name="cart">Cart details.</param>
        internal void AddToCart(Cart cart)
        {
            Cart? existingCart = this._carts.FirstOrDefault(item => item.UserName == cart.UserName && item.ProductId == cart.ProductId);

            if (existingCart != null)
            {
                existingCart.CartQuantity = cart.CartQuantity;
            }
            else
            {
                this._carts.Add(cart);
            }

            this.WriteAll();
        }

        /// <summary>
        /// Removes all cart products of a particular user.
        /// </summary>
        /// <param name="userName">Username of the user.</param>
        internal void RemoveCart(string userName)
        {
            this._carts.RemoveAll(cart => cart.UserName == userName);
            this.WriteAll();
        }

        /// <summary>
        /// Removes selected products from the user's cart.
        /// </summary>
        /// <param name="userName">Username of the user.</param>
        /// <param name="productIds">IDs of products to be removed.</param>
        internal void RemoveCartItems(string userName, List<string> productIds)
        {
            this._carts.RemoveAll(cart => cart.UserName == userName && productIds.Contains(cart.ProductId));
            this.WriteAll();
        }

        private void WriteAll()
        {
            string fileData = JsonSerializer.Serialize(this._carts, this._options);
            File.WriteAllText(this._filePath, fileData);
        }

        private void LoadAll()
        {
            if (!File.Exists(this._filePath))
            {
                this.WriteAll();
                return;
            }

            string fileData = File.ReadAllText(this._filePath);

            if (string.IsNullOrWhiteSpace(fileData))
            {
                return;
            }

            using JsonDocument document = JsonDocument.Parse(fileData);

            foreach (JsonElement item in document.RootElement.EnumerateArray())
            {
                string userName = item.TryGetProperty("UserName", out JsonElement userElement)
                    ? userElement.GetString() ?? string.Empty
                    : string.Empty;

                string productId = item.TryGetProperty("ProductId", out JsonElement idElement)
                    ? idElement.GetString() ?? string.Empty
                    : string.Empty;

                int cartQuantity = item.TryGetProperty("CartQuantity", out JsonElement quantityElement)
                    ? quantityElement.GetInt32()
                    : 0;

                if (string.IsNullOrWhiteSpace(productId) &&
                    item.TryGetProperty("Product", out JsonElement productElement) &&
                    productElement.TryGetProperty("Name", out JsonElement nameElement))
                {
                    string productName = nameElement.GetString() ?? string.Empty;

                    Product? product = this._productRepository.GetAllProducts()
                        .FirstOrDefault(product => product.Name.Equals(productName, StringComparison.OrdinalIgnoreCase));

                    productId = product?.ProductId ?? string.Empty;
                }

                if (!string.IsNullOrWhiteSpace(userName) &&
                    !string.IsNullOrWhiteSpace(productId) &&
                    cartQuantity > 0)
                {
                    this._carts.Add(new Cart
                    {
                        UserName = userName,
                        ProductId = productId,
                        CartQuantity = cartQuantity,
                    });
                }
            }

            this.WriteAll();
        }
    }
}
