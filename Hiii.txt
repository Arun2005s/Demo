
using GroceryShopApp.Model;
using GroceryShopApp.Repository;

namespace GroceryShopApp.Services
{
    internal class ProductServices
    {
        private readonly ProductRepository _productRepository;

        public ProductServices(ProductRepository productRepository)
        {
            this._productRepository = productRepository;
        }

        internal List<Product> GetProductsByCategory(
            Categories category)
        {
            List<Product> products =
                this._productRepository.GetAllProducts();

            return products
                .Where(product => product.Category == category)
                .ToList();
        }

        internal List<Product> SearchProducts(
            Categories category, string searchName)
        {
            List<Product> products =
                this.GetProductsByCategory(category);

            return products
                .Where(product => product.Name.Contains(
                    searchName,
                    StringComparison.OrdinalIgnoreCase))
                .ToList();
        }
    }
}
