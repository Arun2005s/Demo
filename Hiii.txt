
using GroceryShopApp.Model;
using GroceryShopApp.Services;

namespace GroceryShopApp.View
{
    /// <summary>
    /// Performs the view level fetching and display operations for products.
    /// </summary>
    internal class ProductView
    {
        private readonly ProductServices _productServices;
        private readonly CartServices _cartServices;
        private readonly WishlistServices _wishlistServices;
        private readonly CartView _cartView;

        /// <summary>
        /// Initializes a new instance of the <see cref="ProductView"/> class.
        /// </summary>
        /// <param name="productServices">Services of the product.</param>
        /// <param name="cartServices">Services of the cart.</param>
        /// <param name="wishlistServices">Services of the wishlist.</param>
        /// <param name="cartView">View operations of the cart.</param>
        public ProductView(ProductServices productServices, CartServices cartServices, WishlistServices wishlistServices, CartView cartView)
        {
            this._productServices = productServices;
            this._cartServices = cartServices;
            this._wishlistServices = wishlistServices;
            this._cartView = cartView;
        }

        /// <summary>
        /// Displays the product menu and fetches the user choice to redirect.
        /// </summary>
        /// <param name="userName">Username of the current user.</param>
        internal void DisplayProductMenu(string userName)
        {
            while (true)
            {
                Console.Write(@"
==================================
          PRODUCT MENU
==================================
[V]egetables
[F]ruits
[C]ookware
[B]ack to Main Menu

Enter Choice: ");

                ConsoleKey userChoice = Console.ReadKey().Key;
                Console.Clear();

                switch (userChoice)
                {
                    case ConsoleKey.V:
                        this.DisplayProducts(Categories.Vegetable, userName);
                        break;
                    case ConsoleKey.F:
                        this.DisplayProducts(Categories.Fruit, userName);
                        break;
                    case ConsoleKey.C:
                        this.DisplayProducts(Categories.Cookware, userName);
                        break;
                    case ConsoleKey.B:
                        return;
                    default:
                        ConsoleHelper.WriteErrorMessage("\nInvalid choice.\n");
                        break;
                }
            }
        }

        /// <summary>
        /// Displays products belonging to the selected category.
        /// </summary>
        /// <param name="category">Selected product category.</param>
        /// <param name="userName">Username of the current user.</param>
        private void DisplayProducts(Categories category, string userName)
        {
            List<Product> products = this._productServices.GetProductsByCategory(category);

            while (true)
            {
                Console.WriteLine($@"
==================================
            {category.ToString().ToUpper()}
==================================");

                this.DisplayProductList(products);

                Console.Write(@"
[S]earch Product
[B]ack to Categories

Enter product number to select: ");

                string? choice = Console.ReadLine();

                if (string.Equals(choice, "B", StringComparison.OrdinalIgnoreCase))
                {
                    Console.Clear();
                    return;
                }

                if (string.Equals(choice, "S", StringComparison.OrdinalIgnoreCase))
                {
                    Console.Write("\nEnter product name to search: ");
                    string searchName = Console.ReadLine() ?? string.Empty;

                    if (string.IsNullOrWhiteSpace(searchName))
                    {
                        ConsoleHelper.WriteErrorMessage("\nSearch name cannot be empty.\n");
                        continue;
                    }

                    products = this._productServices.SearchProducts(category, searchName.Trim());
                    Console.Clear();

                    if (products.Count == 0)
                    {
                        ConsoleHelper.WriteErrorMessage("\nNo products found.\n");
                    }

                    continue;
                }

                if (!int.TryParse(choice, out int productNumber) || productNumber < 1 || productNumber > products.Count)
                {
                    ConsoleHelper.WriteErrorMessage("\nInvalid product selection.\n");
                    continue;
                }

                Product selectedProduct = products[productNumber - 1];
                this.DisplaySelectedProduct(selectedProduct, userName);
            }
        }

        /// <summary>
        /// Displays the list of products.
        /// </summary>
        /// <param name="products">List of products.</param>
        private void DisplayProductList(List<Product> products)
        {
            if (products.Count == 0)
            {
                ConsoleHelper.WriteErrorMessage("\nNo products found.\n");
                return;
            }

            for (int i = 0; i < products.Count; i++)
            {
                Product product = products[i];

                Console.WriteLine($"\n{i + 1}. Product: {product.Name}");
                Console.WriteLine($"   Price: ₹{product.Price}");
                Console.WriteLine(product.Quantity == 0 ? "   Status: Out of stock" : $"   Remaining Stock: {product.Quantity}");
            }
        }

        /// <summary>
        /// Displays the selected product and performs the user's selected operation.
        /// </summary>
        /// <param name="product">Selected product.</param>
        /// <param name="userName">Username of the current user.</param>
        private void DisplaySelectedProduct(Product product, string userName)
        {
            Console.WriteLine($@"
==================================
        PRODUCT DETAILS
==================================
Name: {product.Name}
Price: ₹{product.Price}
Category: {product.Category}
Stock: {product.Quantity}
Image: {product.ImageFile}");

            if (product.Quantity == 0)
            {
                ConsoleHelper.WriteErrorMessage("\nOut of stock. Cannot purchase this product.\n");
                return;
            }

            Console.Write(@"
[A]dd to Cart
[P]urchase Now
[W]ishlist
[B]ack

Enter Choice: ");

            ConsoleKey choice = Console.ReadKey().Key;
            Console.WriteLine();

            switch (choice)
            {
                case ConsoleKey.A:
                    this.AddProductToCart(product, userName);
                    break;
                case ConsoleKey.P:
                    this.PurchaseProduct(product, userName);
                    break;
                case ConsoleKey.W:
                    string result = this._wishlistServices.AddToWishlist(userName, product.Name);

                    if (result == "Product added to wishlist successfully.")
                    {
                        ConsoleHelper.WriteSuccessMessage($"\n{result}\n");
                    }
                    else
                    {
                        ConsoleHelper.WriteErrorMessage($"\n{result}\n");
                    }

                    break;
                case ConsoleKey.B:
                    return;
                default:
                    ConsoleHelper.WriteErrorMessage("\nInvalid choice.\n");
                    break;
            }
        }

        /// <summary>
        /// Adds the selected product to the user's cart.
        /// </summary>
        /// <param name="product">Selected product.</param>
        /// <param name="userName">Username of the current user.</param>
        private void AddProductToCart(Product product, string userName)
        {
            Console.Write("\nEnter quantity: ");

            if (!int.TryParse(Console.ReadLine(), out int quantity) || quantity <= 0)
            {
                ConsoleHelper.WriteErrorMessage("\nInvalid quantity.\n");
                return;
            }

            string result = this._cartServices.AddToCart(userName, product.Name, quantity);

            if (result == "Product added to cart successfully.")
            {
                ConsoleHelper.WriteSuccessMessage($"\n{result}\n");
            }
            else
            {
                ConsoleHelper.WriteErrorMessage($"\n{result}\n");
            }
        }

        /// <summary>
        /// Purchases the selected product immediately or adds it to the cart.
        /// </summary>
        /// <param name="product">Selected product.</param>
        /// <param name="userName">Username of the current user.</param>
        private void PurchaseProduct(Product product, string userName)
        {
            Console.Write("\nEnter quantity: ");

            if (!int.TryParse(Console.ReadLine(), out int quantity) || quantity <= 0)
            {
                ConsoleHelper.WriteErrorMessage("\nInvalid quantity.\n");
                return;
            }

            Console.Write("\nDo you want to purchase now? [Y/N]: ");
            ConsoleKey confirmation = Console.ReadKey().Key;
            Console.WriteLine();

            string result = this._cartServices.AddToCart(userName, product.Name, quantity);

            if (result != "Product added to cart successfully.")
            {
                ConsoleHelper.WriteErrorMessage($"\n{result}\n");
                return;
            }

            if (confirmation == ConsoleKey.Y)
            {
                this._cartView.Checkout(userName, new List<string> { product.Name });
            }
            else
            {
                ConsoleHelper.WriteSuccessMessage("\nProduct added to cart successfully.\n");
            }
        }
    }
}
