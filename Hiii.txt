
using System.Text.Json;
using System.Text.Json.Serialization;
using GroceryShopApp.Model;

namespace GroceryShopApp.Repository
{
    internal class ProductRepository
    {
        private readonly List<Product> _products =
            new List<Product>();

        private readonly string _filePath;

        private readonly JsonSerializerOptions _options =
            new JsonSerializerOptions
            {
                WriteIndented = true,
                Converters = { new JsonStringEnumConverter() }
            };

        public ProductRepository(string filePath)
        {
            this._filePath = filePath;
            this.LoadAll();
        }

        internal List<Product> GetAllProducts()
        {
            return this._products.ToList();
        }

        private void WriteAll()
        {
            string fileData = JsonSerializer.Serialize(
                this._products, this._options);

            File.WriteAllText(this._filePath, fileData);
        }

        private void LoadAll()
        {
            // Check whether the file exists
            if (!File.Exists(this._filePath))
            {
                this._products.AddRange(this.GetDefaultProducts());

                this.WriteAll();

                return;
            }

            // Read the file contents
            string fileData = File.ReadAllText(this._filePath);

            // Check whether the file is empty
            if (string.IsNullOrWhiteSpace(fileData))
            {
                this._products.AddRange(this.GetDefaultProducts());

                this.WriteAll();

                return;
            }

            // Deserialize JSON into in-memory collection
            List<Product> products =
                JsonSerializer.Deserialize<List<Product>>(
                    fileData, this._options
                ) ?? new List<Product>();

            // Check whether the deserialized collection is empty
            if (products.Count == 0)
            {
                this._products.AddRange(this.GetDefaultProducts());

                this.WriteAll();

                return;
            }

            this._products.AddRange(products);
        }
